    <script>
        // Web Audio API 사운드 신디사이저 효과음 생성기
        let audioCtx = null;
        function playSynthSound(type) {
            try {
                if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                const vol = parseFloat(document.getElementById('volumeSlider').value);

                osc.connect(gain);
                gain.connect(audioCtx.destination);
                gain.gain.setValueAtTime(vol, audioCtx.currentTime);

                if (type === 'click') {
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(600, audioCtx.currentTime);
                    osc.frequency.exponentialRampToValueAtTime(200, audioCtx.currentTime + 0.05);
                    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.05);
                    osc.start();
                    osc.stop(audioCtx.currentTime + 0.05);
                } else if (type === 'shutter') {
                    osc.type = 'square';
                    osc.frequency.setValueAtTime(150, audioCtx.currentTime);
                    osc.frequency.exponentialRampToValueAtTime(40, audioCtx.currentTime + 0.15);
                    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.15);
                    osc.start();
                    osc.stop(audioCtx.currentTime + 0.15);
                } else if (type === 'focus') {
                    osc.type = 'triangle';
                    osc.frequency.setValueAtTime(800, audioCtx.currentTime);
                    osc.frequency.setValueAtTime(1200, audioCtx.currentTime + 0.08);
                    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.15);
                    osc.start();
                    osc.stop(audioCtx.currentTime + 0.15);
                }
            } catch(e) { console.error(e); }
        }

        // 웹캠 연결 초기화
        const video = document.getElementById('webcam');
        async function initCamera() {
            try {
                const stream = await navigator.mediaDevices.getUserMedia({ 
                    video: { width: { ideal: 1920 }, height: { ideal: 1080 } }, 
                    audio: false 
                });
                video.srcObject = stream;
            } catch (err) {
                alert('웹캠 접근 권한이 필요합니다. 브라우저 설정을 확인해주세요.');
            }
        }
        initCamera();

        // 렌즈 선택 및 디지털 줌 로직
        let currentLensScale = 1.0;
        let baseZoom = 1.0;

        function setLens(scale, name, btn) {
            playSynthSound('click');
            document.querySelectorAll('.lens-buttons .synth-btn').forEach(b => b.classList.remove('active'));
            btn.classList.add('active');
            
            baseZoom = scale;
            document.getElementById('zoomSlider').value = scale;
            updateZoom(scale);
            document.getElementById('hudLens').innerText = name;
        }

        function updateZoom(val) {
            currentLensScale = parseFloat(val);
            document.getElementById('zoomVal').innerText = currentLensScale.toFixed(1) + 'x';
            applyVideoTransform();
        }

        // 필터 및 OIS 흔들림 보정 시뮬레이션
        let oisState = true;
        let brightness = 0;

        function toggleOIS() {
            playSynthSound('click');
            oisState = !oisState;
            const btn = document.getElementById('oisBtn');
            const hudOis = document.getElementById('hudOis');
            if (oisState) {
                btn.style.borderColor = 'var(--neon-cyan)';
                hudOis.innerText = 'OIS: ACTIVE';
            } else {
                btn.style.borderColor = '#555';
                hudOis.innerText = 'OIS: OFF';
            }
            applyVideoTransform();
        }

        function toggleGrid() {
            playSynthSound('click');
            const grid = document.getElementById('gridLines');
            const btn = document.getElementById('gridBtn');
            grid.classList.toggle('active');
            btn.style.borderColor = grid.classList.contains('active') ? 'var(--neon-cyan)' : '#555';
        }

        function updateFilters() {
            brightness = document.getElementById('brightSlider').value;
            document.getElementById('brightVal').innerText = brightness;
            applyVideoTransform();
        }

        function applyVideoTransform() {
            let filterStr = `brightness(${100 + parseInt(brightness)}%)`;
            if (oisState) {
                // 부드러운 스무딩 필터 느낌
                filterStr += ` contrast(105%)`;
            }
            video.style.transform = `scale(${currentLensScale})`;
            video.style.filter = filterStr;
        }

        // 터치/클릭 포커스 시뮬레이션
        const viewport = document.getElementById('viewport');
        const focusBox = document.getElementById('focusBox');

        viewport.addEventListener('click', (e) => {
            const rect = viewport.getBoundingClientRect();
            const x = e.clientX - rect.left;
            const y = e.clientY - rect.top;

            focusBox.style.left = `${x}px`;
            focusBox.style.top = `${y}px`;
            focusBox.classList.remove('animate');
            void focusBox.offsetWidth; // reflow
            focusBox.classList.add('animate');

            playSynthSound('focus');
            document.getElementById('hudAf').innerText = `AF: LOCKED (${Math.round(x)},${Math.round(y)})`;
            setTimeout(() => {
                document.getElementById('hudAf').innerText = 'AF: CONTINUOUS';
            }, 1200);
        });

        function triggerAutoFocus() {
            playSynthSound('focus');
            focusBox.style.left = `50%`;
            focusBox.style.top = `50%`;
            focusBox.classList.remove('animate');
            void focusBox.offsetWidth;
            focusBox.classList.add('animate');
        }

        // 스냅샷 촬영 및 갤러리 추가
        function takeSnapshot() {
            playSynthSound('shutter');
            const canvas = document.getElementById('snapshotCanvas');
            canvas.width = video.videoWidth || 1280;
            canvas.height = video.videoHeight || 720;
            const ctx = canvas.getContext('2d');
            
            // 줌 및 필터 반영 캡처
            ctx.save();
            ctx.translate(canvas.width / 2, canvas.height / 2);
            ctx.scale(currentLensScale, currentLensScale);
            ctx.translate(-video.videoWidth / 2, -video.videoHeight / 2);
            ctx.filter = video.style.filter;
            ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
            ctx.restore();

            const dataUrl = canvas.toDataURL('image/png');
            
            // 갤러리 스트립에 추가
            const strip = document.getElementById('galleryStrip');
            const emptyMsg = strip.querySelector('.empty-gallery');
            if (emptyMsg) emptyMsg.remove();

            const img = document.createElement('img');
            img.src = dataUrl;
            img.className = 'gallery-thumb';
            img.onclick = () => {
                const w = window.open('');
                w.document.write(`<img src="${dataUrl}" style="width:100%;background:#000;" />`);
            };
            strip.prepend(img);
        }
